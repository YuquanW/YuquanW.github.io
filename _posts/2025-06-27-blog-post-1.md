# Groovy code for channel outputs:

```
import qupath.lib.regions.RegionRequest
import qupath.lib.images.ImageData
import qupath.lib.objects.PathAnnotationObject
import qupath.lib.awt.common.AwtTools
import qupath.lib.images.servers.TransformedServerBuilder

def imageData = getCurrentImageData()
def server = getCurrentServer()
def annotations = getAnnotationObjects()
if (annotations.isEmpty()) {
    print "Annotation not found!\n"
    return
}
print "Number of annotations: ${annotations.size()}\n"
for (int a = 0; annotations.size(); a++) {
    def annotation = annotations[a]
    def roi = annotation.getROI()
    def path = buildFilePath("./ROI/", "roi"+a)
    mkdirs(path)
    
    def channels = server.getMetadata().getChannels()
    def nChannels = server.nChannels()
    
    for (int c = 0; c < nChannels; c++) {
        def channelName = channels[c].getName()
        def outputPath = buildFilePath(path, channelName + ".tiff")
        print("${c+1}: ${outputPath}\n")

        def singleChannel = new TransformedServerBuilder(server)
            .extractChannels(c)
            .build()
        def request = RegionRequest.createInstance(server.getPath(), 2.2, roi)
        def roiImage = singleChannel.readRegion(request)
    
        def outputFile = new File(outputPath)
        writeImage(roiImage,outputPath)
    }
    print "Annotation${a+1} finished.\n"
//    def outputPath = buildFilePath(path, "roi"+a+".tiff")
//    writeImageRegion(server, request, outputPath)
}
```
