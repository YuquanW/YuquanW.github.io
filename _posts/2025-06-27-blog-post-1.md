# Groovy code for rectangle grid annotation:

```
import qupath.lib.objects.PathAnnotationObject
import qupath.lib.roi.RectangleROI
import qupath.lib.common.GeneralTools
import qupath.lib.images.ImageData
import qupath.lib.regions.RegionRequest

int nRows = 17                    // Row numbers
int nCols = 9                   // Col numbers
double coreWidthMm = 1.5         // Annotation rectangle width
double coreHeightMm = 1.5        // Annotation rectangle height
// --------------------------------------------------------------

def imageData = getCurrentImageData()
def server = imageData.getServer()

// ---------------------pixel <--> microns------------------------
def calibration = server.getPixelCalibration()    
double pixelWidthUm = calibration.getPixelWidthMicrons()   
double pixelHeightUm = calibration.getPixelHeightMicrons() 

double imageWidthPx = server.getWidth()
double imageHeightPx = server.getHeight()

double imageWidthUm = imageWidthPx * pixelWidthUm
double imageHeightUm = imageHeightPx * pixelHeightUm

double coreWidthUm  = coreWidthMm  * 1000.0
double coreHeightUm = coreHeightMm * 1000.0

// --------------------Gap size calc---------------------------
// (nCols * coreWidthUm + (nCols - 1) * dxUm) = imageWidthUm
// => dxUm = (imageWidthUm - nCols * coreWidthUm) / (nCols - 1)

double dxUm = (imageWidthUm - nCols * coreWidthUm) / (nCols + 1)
double dyUm = (imageHeightUm - nRows * coreHeightUm) / (nRows + 1)

// --------------------Create annotation--------------------------
int count = 0
def letters = 'ABCDEFGHIJKLMNOPQRSTUVWXYZ'
for (int row = 0; row < nRows; row++) {
    for (int col = 0; col < nCols; col++) {
        double xUm = dxUm + col * (coreWidthUm + dxUm)
        double yUm = dyUm + row * (coreHeightUm + dyUm)
        
        def roi = new RectangleROI(xUm/calibration.getPixelWidthMicrons(), yUm/calibration.getPixelHeightMicrons(), 
                                   coreWidthUm/calibration.getPixelWidthMicrons(), coreHeightUm/calibration.getPixelHeightMicrons())
        def anno = new PathAnnotationObject(roi)
        
        char colLetter = letters.charAt(nCols - col)
        anno.setName("${colLetter}-${nRows+1-row}")
        imageData.getHierarchy().addObject(anno, false)
        count++
    }
}
```
